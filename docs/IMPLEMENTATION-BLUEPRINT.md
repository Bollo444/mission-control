# PART A — Live-LLM Team Meeting

**Goal:** Each agent's meeting turn is generated by a real `cascadeChat()` call to its routed model, with the agent's persona as the system prompt and live fleet metrics as context. Keeps the existing topic/thread/roster structure; replaces the canned `PERSONAS[id].*` string functions with LLM generation.

**Architecture:** Keep `buildMeeting(report)`'s orchestration (convene → topics → check-ins → close). For each turn, instead of `persona.status(c)` returning a template, call a new `llmTurn(agentId, phase, ctx, prompt)` that builds an OpenAI chat request with persona-as-system-prompt + metric context, runs it through `cascadeChat`, and returns the text. Falls back to the existing template if the call fails (so the meeting never breaks).

**Why fall back:** The meeting is a "nice to have" view — it must never 503 the dashboard. Every LLM turn wraps in try/catch → template fallback.

  report: SystemReport;
  phase: "open" | "reply" | "close";
  topic?: string;          // human label of the current thread, if any
  priorTurns?: string[];   // recent lines already said in this thread (so agents reply to each other)
  userMessage?: string;    // set when replying to a user-typed message
}

/** Build the system prompt that defines an agent's voice + lens. */
function personaPrompt(agentId: string): string {
  const a = getAgent(agentId);
  const beh = getAgentBehavior(agentId);
  const role = beh.role ?? a?.tagline ?? "Agent";
  const lens = beh.lens ?? "your specialty";
  return [
    `You are ${a?.name ?? agentId}, the ${role} in a team of AI coding agents.`,
    `Your lens: ${lens}. Speak only to what you genuinely own.`,
    "Rules: 1-3 sentences. Conversational, first person, in character. Reference real numbers from the context.",
    "Address other agents by name when replying to them. No markdown, no headers, no bullet lists — just speech.",
  ].join(" ");
}

/** Render the live fleet metrics as compact context the model can cite. */
function metricsBlock(report: SystemReport): string {
  const f = report.fleet;
  const agents = f.agents
    .map((x) => `${x.name}=${x.state}`)
    .join(", ");
  return [
    `FLEET: ${f.ready}/${f.total} live · ${f.config} config · ${f.offline} offline · ${f.sessions} sessions.`,
    `Agents: ${agents}`,
    `HOST: CPU ${report.cpu.usagePct}% · mem ${report.memory.usedPct}%` +
      (report.disk ? ` · disk ${report.disk.usedPct}%` : "") +
      ` · ${report.cpu.cores} cores.`,
    `VAULT: ${report.vault.agentNotes} notes · ${report.vault.activityEntries} activity entries` +
      (report.vault.lastActivity ? ` · last ${report.vault.lastActivity}` : " · no activity yet") + ".",
  ].join("\n");

```typescript
/**
 * Generate one agent turn via cascadeChat on the agent's routed model.
 * Returns the spoken text, or null if the call failed (caller falls back).
 */
export async function llmTurn(agentId: string, ctx: TurnCtx): Promise<string | null> {
  const settings = readSettings();
  const route = settings.routingPreferred[agentId] ?? settings.routing[agentId];
  if (!route) return null;

  const userParts: string[] = [];
  if (ctx.phase === "open") {
    userParts.push("Open the meeting: frame the agenda from the live state in 1-2 sentences.");
  } else if (ctx.phase === "close") {
    userParts.push("Close the meeting: land the decision crisply in 1-2 sentences. Summarize, don't re-run.");
  } else {
    if (ctx.userMessage) userParts.push(`The user said: "${ctx.userMessage}". Respond in character.`);
    if (ctx.topic) userParts.push(`Current thread: ${ctx.topic}.`);
    if (ctx.priorTurns && ctx.priorTurns.length) {
      userParts.push("Recent lines in this thread:\n" + ctx.priorTurns.map((t) => `- ${t}`).join("\n"));
      userParts.push("Reply to these — address the last speaker by name.");
    }
  }
  userParts.push("\nLIVE CONTEXT:\n" + metricsBlock(ctx.report));

  const body = {
    model: `${route.provider}/${route.model}`,   // cascadeChat resolves provider/model from this
    messages: [
      { role: "system", content: personaPrompt(agentId) },
      { role: "user", content: userParts.join("\n\n") },
    ],
    max_tokens: 220,
    temperature: 0.8,
    stream: false,
  };

  const result = await cascadeChat(body, { agentId });
  if (!result.ok) {
    logEvent({ source: "meeting", level: "warn", event: "llm turn failed", detail: `${agentId}: ${result.error}`, meta: { agentId } });
    return null;
  }
  const text = await result.response.text().catch(() => "");
  try {
    const json = JSON.parse(text);
    const out = json?.choices?.[0]?.message?.content?.toString().trim();
    return out && out.length > 0 ? out : null;
  } catch {
    return null;
  }
}
```

/**
 * Live-LLM version of the meeting. Same orchestration as buildMeetingTemplated,
 * but each turn is generated through cascadeChat, falling back to the template
 * when a provider call fails. Async because it makes real network calls.
 */
export async function buildMeeting(report: SystemReport): Promise<MeetingResp> {
  const templated = buildMeetingTemplated(report); // produce the structure (turns/roster/metrics)
  // Walk the turns in order; regenerate each one's text via the LLM, keeping
  // the template text as fallback. Thread context: accumulate prior turns per phase.
  const threadBuffer: string[] = [];
  const currentTopic = templated.turns.length ? "fleet standup" : undefined;
  for (const t of templated.turns) {
    let live: string | null = null;
    try {
      live = await llmTurn(t.agentId, {
        report,
        phase: t.phase === "open" ? "open" : t.phase === "close" ? "close" : "reply",
        topic: currentTopic,
        priorTurns: t.phase === "reply" ? threadBuffer.slice(-4) : undefined,
      });
    } catch {
      live = null;
    }
    t.text = live ?? t.text;          // fallback to the template string
    threadBuffer.push(`${t.name}: ${t.text}`);
  }
  return templated;
}

export async function replyToMessage(report: SystemReport, message: string): Promise<MeetingTurn[]> {
  const c = deriveCtx(report);
  __tid = Date.now() % 100000;
  const lower = message.toLowerCase();

  const scored = ORDER.filter((id) => id !== "claude" && id !== "jcode")
    .map((id) => ({ id, score: PERSONAS[id].keywords.reduce((n, k) => (lower.includes(k) ? n + 1 : n), 0) }))
    .filter((x) => x.score > 0)
    .sort((a, b) => b.score - a.score);

  const ids: string[] = scored.length === 0
    ? ["claude", "jcode", ...pickGeneralists(1)]
    : [...scored.slice(0, 3).map((x) => x.id), "jcode", "claude"];

  const turns: MeetingTurn[] = [];
  const buffer: string[] = [];
  for (const id of ids) {
    let text: string | null = null;
    try {
      text = await llmTurn(id, { report, phase: "reply", userMessage: message, priorTurns: buffer.slice(-3) });
    } catch { text = null; }
    const final = text ?? PERSONAS[id].respond(c, message);   // fallback
    buffer.push(`${meta(id).name}: ${final}`);
    turns.push(turn(id, "reply", final));
  }
  return turns;
}

# PART B — Claude Anthropic Bridge + 3-Slot Model Router

**Goal:** Claude Code speaks Anthropic `/v1/messages` to a new endpoint inside Mission Control. That endpoint reads Claude's model selector (haiku/sonnet/opus) and routes to a **3-slot map** you edit from the Settings dashboard — each slot maps to any provider/model in the catalog. Reuses `cascadeChat()` for all provider logic.

**Architecture:**
```
Claude Code ─POST /v1/messages (model:"sonnet[1m]")─▶ :4317 /api/anthropic/v1/messages
   1. auth via gatewayToken
   2. parseSlot(model) → "sonnet" → settings.anthropicSlots.sonnet = {provider,model}
   3. anthropicToOpenAI(req) + inject slot's provider/model
   4. cascadeChat()  (existing 10-provider failover)
   5. stream? streamAdapter() : openAIToAnthropic()
```

/* Anthropic <-> OpenAI translation for the /api/anthropic/v1 bridge.
   Pure functions, fully unit-testable. No I/O, no fetch here. */

export type Slot = "haiku" | "sonnet" | "opus";

/** Map a Claude model string to one of the 3 slots. */
export function parseSlot(model: string | undefined | null): Slot {
  const m = (model ?? "").toLowerCase();
  if (m.includes("haiku")) return "haiku";
  if (m.includes("opus")) return "opus";
  return "sonnet"; // covers "sonnet", "sonnet[1m]", "", null, unknown
}

export interface AnthropicError {
  type: "error";
  error: { type: string; message: string };
}
export function makeAnthropicError(
  status: number,
  message: string
): { status: number; body: AnthropicError } {
  const type =
    status === 401 ? "authentication_error"
    : status === 403 ? "permission_error"
    : status === 404 ? "not_found_error"
    : status === 429 ? "rate_limit_error"
    : status === 529 ? "overloaded_error"
    : status >= 500 ? "api_error"
    : "invalid_request_error";
  return { status, body: { type: "error", error: { type, message } } };
}

- [ ] **Step 2: Request translation (Anthropic → OpenAI)**

```typescript
type AnyObj = Record<string, unknown>;

/** Convert an Anthropic /v1/messages request into an OpenAI /v1/chat/completions body. */
export function anthropicToOpenAI(req: AnyObj): AnyObj {
  const out: AnyObj = { stream: req.stream === true };
  if (typeof req.max_tokens === "number") out.max_tokens = req.max_tokens;
  if (typeof req.temperature === "number") out.temperature = req.temperature;
  if (typeof req.top_p === "number") out.top_p = req.top_p;
  if (Array.isArray(req.stop_sequences)) out.stop = req.stop_sequences;

  const messages: AnyObj[] = [];

  // System: Anthropic takes it top-level (string OR [{type:"text",text}]).
  // OpenAI wants it as the first {role:"system"} message. Strip cache_control.
  if (req.system != null) {
    const sysText = typeof req.system === "string"
      ? req.system
      : (req.system as AnyObj[])
          .filter((b) => b.type === "text")
          .map((b) => b.text)
          .join("\n\n");
    if (sysText) messages.push({ role: "system", content: sysText });
  }

  // Messages: translate each turn's content blocks.
  for (const m of (req.messages as AnyObj[]) ?? []) {
    const role = m.role === "assistant" ? "assistant" : "user";
    const content = m.content;
    if (typeof content === "string") {
      messages.push({ role, content });
      continue;
    }
    // Array of blocks.
    if (role === "assistant") {
      const textParts: string[] = [];
      const toolCalls: AnyObj[] = [];
      for (const b of content as AnyObj[]) {
        if (b.type === "text") textParts.push(b.text);
        else if (b.type === "tool_use") {
          toolCalls.push({
            id: b.id,
            type: "function",
            function: { name: b.name, arguments: JSON.stringify(b.input ?? {}) },
          });
        }
      }
      const msg: AnyObj = { role };
      if (textParts.length) msg.content = textParts.join("");
      if (toolCalls.length) msg.tool_calls = toolCalls;
      messages.push(msg);
    } else {
      // user: text + image + tool_result blocks.
      const parts: AnyObj[] = [];
      const toolResults: AnyObj[] = [];
      for (const b of content as AnyObj[]) {
        if (b.type === "text") parts.push({ type: "text", text: b.text });
        else if (b.type === "image") {
          const src = b.source ?? {};
          parts.push({ type: "image_url", image_url: { url: src.data ? `data:${src.media_type ?? "image/png"};base64,${src.data}` : "" } });
        } else if (b.type === "tool_result") {
          const c = b.content;
          const text = typeof c === "string" ? c : Array.isArray(c) ? (c as AnyObj[]).filter((x) => x.type === "text").map((x) => x.text).join("") : "";
          toolResults.push({ role: "tool", tool_call_id: b.tool_use_id, content: text });
        }
      }
      if (toolResults.length) messages.push(...toolResults);
      if (parts.length) messages.push({ role, content: parts.length === 1 && parts[0].type === "text" ? parts[0].text : parts });
    }
  }

  out.messages = messages;

  // Tools.
  if (Array.isArray(req.tools) && req.tools.length) {
    out.tools = (req.tools as AnyObj[]).map((t) => ({
      type: "function",
      function: {
        name: t.name,
        description: t.description,
        parameters: t.input_schema ?? { type: "object", properties: {} },
      },
    }));
  }
  if (req.tool_choice) {
    const tc = req.tool_choice as AnyObj;
    if (tc.type === "any") out.tool_choice = "required";
    else if (tc.type === "auto") out.tool_choice = "auto";
    else if (tc.type === "tool") out.tool_choice = { type: "function", function: { name: tc.name } };
    else if (tc.type === "none") out.tool_choice = "none";
  }
  return out;
}
```

const STOP_MAP: Record<string, string> = {
  stop: "end_turn",
  length: "max_tokens",
  tool_calls: "tool_use",
  function_call: "tool_use",
};

/** Convert an OpenAI chat completion JSON into an Anthropic /v1/messages response. */
export function openAIToAnthropic(openai: AnyObj, requestedModel: string): AnyObj {
  const choice = (openai.choices as AnyObj[])?.[0] ?? {};
  const msg = (choice.message ?? {}) as AnyObj;
  const finish = String(choice.finish_reason ?? "stop");
  const stopReason = STOP_MAP[finish] ?? "end_turn";

  const content: AnyObj[] = [];
  if (typeof msg.content === "string" && msg.content.length) {
    content.push({ type: "text", text: msg.content });
  }
  if (Array.isArray(msg.tool_calls)) {
    for (const tc of msg.tool_calls as AnyObj[]) {
      let input: unknown = {};
      try { input = JSON.parse(String(tc.function?.arguments ?? "{}")); } catch { input = {}; }
      content.push({
        type: "tool_use",
        id: String(tc.id ?? `toolu_${crypto.randomBytes(12).toString("hex")}`),
        name: String(tc.function?.name ?? ""),
        input,
      });
    }
  }

  const u = (openai.usage ?? {}) as AnyObj;
  return {
    id: `msg_${crypto.randomBytes(12).toString("hex")}`,
    type: "message",
    role: "assistant",
    content: content.length ? content : [{ type: "text", text: "" }],
    model: requestedModel,
    stop_reason: stopReason,
    stop_sequence: null,
    usage: {
      input_tokens: Number(u.prompt_tokens ?? 0),
      output_tokens: Number(u.completion_tokens ?? 0),
      cache_creation_input_tokens: 0,
      cache_read_input_tokens: 0,
    },
  };
}

- [ ] **Step 4: SSE stream re-encoder (the hard part)**

Append a function that takes the upstream OpenAI SSE stream and re-emits the Anthropic event grammar. Returns a `ReadableStream<Uint8Array>`:

```typescript
function sse(event: string, data: unknown): string {
  return `event: ${event}\ndata: ${JSON.stringify(data)}\n\n`;
}

/** Re-encode an OpenAI streaming response as an Anthropic SSE event stream. */
export function streamAdapter(upstreamBody: ReadableStream<Uint8Array>, requestedModel: string): ReadableStream<Uint8Array> {
  const encoder = new TextEncoder();
  const decoder = new TextDecoder();
  let buf = "";
  let blockOpen = false;       // is a text content_block currently open?
  let toolBlockOpen = false;
  let outputTokens = 0;
  let msgId = `msg_${crypto.randomBytes(12).toString("hex")}`;

  return new ReadableStream({
    async start(controller) {
      // message_start first.
      controller.enqueue(encoder.encode(sse("message_start", {
        type: "message_start",
        message: { id: msgId, type: "message", role: "assistant", content: [], model: requestedModel, stop_reason: null, stop_sequence: null, usage: { input_tokens: 0, output_tokens: 0 } },
      })));

      const reader = upstreamBody.getReader();
      const finish = async (stopReason: string) => {
        if (blockOpen) controller.enqueue(encoder.encode(sse("content_block_stop", { type: "content_block_stop", index: 0 })));
        if (toolBlockOpen) controller.enqueue(encoder.encode(sse("content_block_stop", { type: "content_block_stop", index: toolBlockOpen ? 1 : 0 })));
        controller.enqueue(encoder.encode(sse("message_delta", { type: "message_delta", delta: { stop_reason: stopReason, stop_sequence: null }, usage: { output_tokens: outputTokens } })));
        controller.enqueue(encoder.encode(sse("message_stop", { type: "message_stop" })));
        controller.close();
      };
      let stopped = false;
      try {
        for (;;) {
          const { done, value } = await reader.read();
          if (done) break;
          buf += decoder.decode(value, { stream: true });
          const lines = buf.split("\n");
          buf = lines.pop() ?? "";
          for (const line of lines) {
            const trimmed = line.trim();
            if (!trimmed.startsWith("data:")) continue;
            const payload = trimmed.slice(5).trim();
            if (payload === "[DONE]") { if (!stopped) { stopped = true; await finish("end_turn"); } continue; }
            let json: AnyObj;
            try { json = JSON.parse(payload); } catch { continue; }
            const choice = (json.choices as AnyObj[])?.[0];
            if (!choice) continue;
            const delta = (choice.delta ?? {}) as AnyObj;
            // Text delta.
            if (typeof delta.content === "string" && delta.content) {
              if (!blockOpen) {
                blockOpen = true;
                controller.enqueue(encoder.encode(sse("content_block_start", { type: "content_block_start", index: 0, content_block: { type: "text", text: "" } })));
              }
              outputTokens += 1; // rough
              controller.enqueue(encoder.encode(sse("content_block_delta", { type: "content_block_delta", index: 0, delta: { type: "text_delta", text: delta.content } })));
            }
            // Tool call delta (OpenAI streams arguments in fragments).
            if (Array.isArray(delta.tool_calls)) {
              for (const tc of delta.tool_calls as AnyObj[]) {
                if (tc.id && !toolBlockOpen) {
                  toolBlockOpen = true;
                  controller.enqueue(encoder.encode(sse("content_block_start", { type: "content_block_start", index: 1, content_block: { type: "tool_use", id: tc.id, name: tc.function?.name ?? "", input: {} } })));
                }
                const frag = tc.function?.arguments;
                if (typeof frag === "string" && frag) {
                  outputTokens += 1;
                  controller.enqueue(encoder.encode(sse("content_block_delta", { type: "content_block_delta", index: 1, delta: { type: "input_json_delta", partial_json: frag } })));
                }
              }
            }
            if (choice.finish_reason && !stopped) {
              stopped = true;
              await finish(STOP_MAP[String(choice.finish_reason)] ?? "end_turn");
            }
          }
        }
      } catch {
        if (!stopped) { stopped = true; await finish("end_turn"); }
      }
      if (!stopped) await finish("end_turn");
    },
  });
}
```

```typescript
import { cascadeChat } from "@/lib/gateway";
import { getGatewayToken, readSettings } from "@/lib/settings";
import { recordTokens } from "@/lib/usage";
import { logEvent } from "@/lib/logbook";
import {
  anthropicToOpenAI,
  openAIToAnthropic,
  streamAdapter,
  parseSlot,
  makeAnthropicError,
} from "@/lib/anthropic-bridge";

export const runtime = "nodejs";
export const dynamic = "force-dynamic";

/* Claude Code / Anthropic-SDK bridge. POST /api/anthropic/v1/messages
   speaks the Anthropic Messages API; routes to the slot's provider/model
   via cascadeChat, translating shapes both ways. */

function authorized(req: Request): boolean {
  const provided = (req.headers.get("authorization") || "").replace(/^Bearer\s+/i, "").trim();
  const xkey = (req.headers.get("x-api-key") || "").trim();
  const tok = getGatewayToken();
  return Boolean(provided && provided === tok) || Boolean(xkey && xkey === tok);
}

export async function POST(req: Request, { params }: { params: Promise<{ path: string[] }> }) {
  const { path } = await params;
  const sub = (path ?? []).join("/");

  if (!authorized(req)) {
    const e = makeAnthropicError(401, "Unauthorized — use your Mission Control gateway token.");
    return Response.json(e.body, { status: e.status });
  }
  if (!sub.endsWith("messages")) {
    const e = makeAnthropicError(404, `Unsupported path: ${sub}`);
    return Response.json(e.body, { status: e.status });
  }

  const body = (await req.json().catch(() => ({}))) as Record<string, unknown>;
  const wantsStream = body.stream === true;
  const slot = parseSlot(body.model as string | undefined);
  const slots = readSettings().anthropicSlots;
  const route = slots?.[slot];

  const translated = anthropicToOpenAI(body);
  // Force the slot's provider/model into the cascade body.
  if (route) translated.model = `${route.provider}/${route.model}`;

  const result = await cascadeChat(translated, {});
  if (!result.ok) {
    const e = makeAnthropicError(result.status === 429 ? 429 : 503, result.error);
    return Response.json(e.body, { status: e.status });
  }

  const upstream = result.response;
  const servedModel = `claude-${slot}`; // what Claude thinks it got

  if (!wantsStream) {
    const text = await upstream.text();
    let openaiJson: Record<string, unknown>;
    try { openaiJson = JSON.parse(text); } catch { openaiJson = { choices: [{ message: { content: text } }] }; }
    const tok = (openaiJson.usage as { total_tokens?: number } | undefined)?.total_tokens;
    if (typeof tok === "number") recordTokens(result.served.provider, tok);
    logEvent({ source: "anthropic-bridge", level: "info", event: "served", detail: `${slot} → ${result.served.provider}/${result.served.model}`, meta: { slot } });
    return Response.json(openAIToAnthropic(openaiJson, servedModel), {
      headers: { "X-MC-Served-By": `${result.served.provider}/${result.served.model}` },
    });
  }

  // Streaming: re-encode upstream SSE as Anthropic events.
  if (upstream.body) {
    logEvent({ source: "anthropic-bridge", level: "info", event: "streamed", detail: `${slot} → ${result.served.provider}/${result.served.model}`, meta: { slot } });
    return new Response(streamAdapter(upstream.body, servedModel), {
      headers: {
        "Content-Type": "text/event-stream",
        "Cache-Control": "no-cache",
        "X-MC-Served-By": `${result.served.provider}/${result.served.model}`,
      },
    });
  }
  const e = makeAnthropicError(503, "Upstream returned no body.");
  return Response.json(e.body, { status: e.status });
}

export async function GET(req: Request, { params }: { params: Promise<{ path: string[] }> }) {
  const { path } = await params;
  if (!authorized(req)) {
    const e = makeAnthropicError(401, "Unauthorized.");
    return Response.json(e.body, { status: e.status });
  }
  if ((path ?? []).join("/").endsWith("models")) {
    return Response.json({
      object: "list",
      data: [
        { id: "claude-haiku-4-5", object: "model", created: 0, owned_by: "mission-control" },
        { id: "claude-sonnet-4-6", object: "model", created: 0, owned_by: "mission-control" },
        { id: "claude-opus-4-8", object: "model", created: 0, owned_by: "mission-control" },
      ],
    });
  }
  const e = makeAnthropicError(404, "Not found");
  return Response.json(e.body, { status: e.status });
}
```
